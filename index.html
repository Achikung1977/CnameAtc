หากยังไม่ได้ตั้งค่า Web App สามารถใช้งานโหมดข้อมูลจำลองได้ทันที

คัดลอกโค้ดด้านล่างนี้ ไปวางใน Google Sheet -> ส่วนขยาย (Extensions) -> Apps Script แล้วกด การทำให้ใช้งานได้ (Deploy) -> การทำให้ใช้งานได้ใหม่ (New deployment) -> เลือก Web App (ทุกคน/Anyone)

// ==========================================
// Google Apps Script backend for Attendance Web App
// Sheet ID: 1JaRmJrsninnCj6DmlysrnXeSghOUjPgg3ZLN0znqULc
// ==========================================

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dataSheet = ss.getSheetByName("data");
  var actSheet = ss.getSheetByName("act");
  
  // Read activity title from 'act' sheet A1:A2
  var actTitle = "กิจกรรมเข้าร่วม";
  var actSub = "";
  if (actSheet) {
    actTitle = actSheet.getRange("A1").getValue() || actTitle;
    actSub = actSheet.getRange("A2").getValue() || "";
  }
  
  // Read student data from 'data' sheet
  var studentData = [];
  if (dataSheet) {
    var values = dataSheet.getDataRange().getValues();
    // Header is row 0: รหัสนักศึกษา | คำนำหน้า | ชื่อ | นามสกุล | เลขที่ | ระดับชั้น | Check | หมายเหตุ
    for (var i = 1; i < values.length; i++) {
      var row = values[i];
      if (row[0]) { // Check if student ID exists
        studentData.push({
          id: String(row[0]),
          prefix: String(row[1] || ""),
          firstName: String(row[2] || ""),
          lastName: String(row[3] || ""),
          no: String(row[4] || ""),
          classLevel: String(row[5] || ""),
          check: String(row[6] || ""),
          remark: String(row[7] || ""),
          rowIndex: i + 1
        });
      }
    }
  }
  
  var result = {
    eventTitle: actTitle,
    eventSub: actSub,
    students: studentData
  };
  
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var dataSheet = ss.getSheetByName("data");
    var params = JSON.parse(e.postData.contents);
    var action = params.action;
    
    var values = dataSheet.getDataRange().getValues();
    var targetRow = -1;
    
    // Find row by student ID (Col A)
    for (var i = 1; i < values.length; i++) {
      if (String(values[i][0]) === String(params.studentId)) {
        targetRow = i + 1;
        break;
      }
    }
    
    if (targetRow !== -1) {
      if (action === "updateAttendance") {
        // Col G = Check (7), Col H = Remark (8)
        dataSheet.getRange(targetRow, 7).setValue(params.check);
        dataSheet.getRange(targetRow, 8).setValue(params.remark);
      } else if (action === "updateInfo") {
        // Col B = Prefix (2), Col C = First (3), Col D = Last (4), Col F = Class (6)
        dataSheet.getRange(targetRow, 2).setValue(params.prefix);
        dataSheet.getRange(targetRow, 3).setValue(params.firstName);
        dataSheet.getRange(targetRow, 4).setValue(params.lastName);
        dataSheet.getRange(targetRow, 6).setValue(params.classLevel);
      }
      
      return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Student not found" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
