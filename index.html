// Google Apps Script backend for Spreadsheet ID: 1JaRmJrsninnCj6DmlysrnXeSghOUjPgg3ZLN0znqULc
const SPREADSHEET_ID = "1JaRmJrsninnCj6DmlysrnXeSghOUjPgg3ZLN0znqULc";

function doGet(e) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let eventName = "กิจกรรมเช็คชื่อเข้าร่วม";
    const actSheet = ss.getSheetByName("act");
    if (actSheet) {
      const actData = actSheet.getRange("A1:A2").getValues();
      if (actData[0] && actData[0][0]) eventName = actData[0][0];
      if (actData[1] && actData[1][0]) eventName += " (" + actData[1][0] + ")";
    }

    const dataSheet = ss.getSheetByName("data");
    if (!dataSheet) return createJsonResponse({ status: "error", message: "ไม่พบ Sheet ชื่อ 'data'" });

    const rows = dataSheet.getDataRange().getValues();
    const students = [];
    for (let i = 1; i < rows.length; i++) {
      const row = rows[i];
      if (!row[0]) continue;
      const checkVal = String(row[7] || "").toLowerCase().trim(); // Col H
      students.push({
        rowIndex: i + 1,
        studentId: String(row[0] || ""),     // Col A
        prefix: String(row[1] || ""),        // Col B
        firstName: String(row[2] || ""),     // Col C
        lastName: String(row[3] || ""),      // Col D
        no: String(row[4] || ""),            // Col E
        phone: String(row[5] || ""),         // Col F (เบอร์โทรนักศึกษา)
        classLevel: String(row[6] || "ไม่ระบุ"), // Col G (ระดับชั้น)
        check: checkVal === "true" || checkVal === "checked" || checkVal === "1" || checkVal === "เข้าร่วม", // Col H
        note: String(row[8] || ""),          // Col I (หมายเหตุ)
        editedFields: []
      });
    }
    return createJsonResponse({ status: "success", eventName: eventName, data: students });
  } catch (err) {
    return createJsonResponse({ status: "error", message: err.toString() });
  }
}

function doPost(e) {
  try {
    const contents = JSON.parse(e.postData.contents);
    const action = contents.action;
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const dataSheet = ss.getSheetByName("data");

    if (action === "checkAttendance") {
      const rowIdx = findRowByStudentId(dataSheet, String(contents.studentId));
      if (rowIdx !== -1) {
        dataSheet.getRange(rowIdx, 8).setValue(contents.check ? "เข้าร่วม" : ""); // Col H
        dataSheet.getRange(rowIdx, 9).setValue(contents.note || ""); // Col I
        return createJsonResponse({ status: "success" });
      }
    } else if (action === "updateStudentInfo") {
      const rowIdx = findRowByStudentId(dataSheet, String(contents.studentId));
      if (rowIdx !== -1) {
        dataSheet.getRange(rowIdx, 2).setValue(contents.prefix);    // Col B
        dataSheet.getRange(rowIdx, 3).setValue(contents.firstName); // Col C
        dataSheet.getRange(rowIdx, 4).setValue(contents.lastName);  // Col D
        dataSheet.getRange(rowIdx, 6).setValue(contents.phone);     // Col F
        dataSheet.getRange(rowIdx, 7).setValue(contents.classLevel);// Col G
        return createJsonResponse({ status: "success" });
      }
    }
    return createJsonResponse({ status: "error", message: "Invalid action" });
  } catch (err) {
    return createJsonResponse({ status: "error", message: err.toString() });
  }
}

function findRowByStudentId(sheet, studentId) {
  const data = sheet.getRange("A:A").getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]) === studentId) return i + 1;
  }
  return -1;
}

function createJsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}
